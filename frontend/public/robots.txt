# ============================================
# robots.txt - 搜索引擎爬虫协议
# ============================================
# 作用：告诉搜索引擎哪些页面可以抓取
# 位置：必须放在网站根目录（/robots.txt）
# ============================================

# 允许所有爬虫访问所有内容
User-agent: *

# 允许抓取的目录（留空表示允许所有）
Allow: /

# 禁止抓取的路径
# - 后台管理页面（无需索引）
# - API 接口（无需索引）
Disallow: /admin/
Disallow: /login
Disallow: /api/

# 站点地图位置（帮助搜索引擎快速发现所有页面）
Sitemap: /api/sitemap.xml
